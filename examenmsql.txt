consultas
select producto.nombre, producto_precio, categoria.nombre AS categoria
FROM producto 
Join categoria on producto.categoria_fk = categoria.id

Select venta.id, venta.fecha,
	concat (cliente.nombre, ' ', cliente.apellido) AS cliente,
    concat (empleado.nombre, ' ', empleado,apellido) AS empleado
FROM venta
JOIN cliente ON venta.cliente_fk = cliente.id
JOIN empleado ON venta.empleado_fk = empleado.id;

SELECT categoria.nombre, count(producto.id) AS cantidad_productos
FROM categoria
JOIN producto ON producto.categoria_fk = categoria.id
group by categoria.id
ORDER BY cantidad_productos DESC;

select producto.nombre, producto.precio, categoria.nombre AS categoria
FROM producto
JOIN categoria ON producto.categoria_fk = categoria.id
left join detalle_venta ON detalle_venta.producto_fk = producto.id
where detalle_venta.id is null;
